scripted_trigger byz_ruled_by_palaiologos_trigger = {#check if country ruled by a Palaiologos; shamelessly stolen from BRANDENBURG EVENTS
	has_ruler = yes
	ruler.dynasty ?= dynasty:palaiologos_dynasty
	OR = {
		government_type = government_type:monarchy
		government_type = government_type:admin_imperial
	}
}

scripted_trigger byz_has_dynastic_chaos_active_trigger = {
	has_country_modifier = byz_imperial_neglect
}

scripted_effect byz_change_control_score = {
	custom_description = {
		text = byz_change_control_score_$math$_text
		#text = bra_change_investment_score_$math$_text
		value = $value$
		change_variable = {
			name = pm_byz_imperial_authority_var
			$math$ = $value$
		}
	}
}

scripted_effect byz_set_localization_scopes = {
	save_scope_as = root_country
	ruler ?= { 
		save_scope_as = root_ruler 
		dynasty ?= { save_scope_as = root_dynasty }
	}
}

#Twilight of the Palaiologos
#	High-level Concept: Represents the military (and related admistrative) challenges associated with the decline of the Byzantine Empire between 1336 and 1453
#	Duration: 100 years OR satisfaction conditions met (Imperial Authority restored, Dynasty Swap, Government Type Swap)
#	Mechanics: Imperial Authority Events
#	How does it work: 
#		* Events that increase or decrease "Imperial Authority" variable. 
#		* Based on the "Imperial Authority" variable, maybe at different 'levels' different events can happen
#			+ Ex: "Soldiers Acclaim" Event has no level and two possibilities; Positive: "Soldiers Acclaim the Basileus" just gives you a bonus and increments "Imperial Authority", Negative "Soldiers Acclaim the Strategos" makes the player choose between losing Imperial Authority and legitimacy or spending stability negate the "Imperial Authority loss (logic of the event = choose a random location in the country for the scope, then add either the "endorses the basileus" (soldier pops happy) or "endorses the strategoi" (soldier pops unhappy) modifier
#			+ Ex: "Unhappy Generals", players must either spend gold and increase legitimacy or lose stability and decrease Imperial Authority 
#			+ Ex: "Governor conspires with foreign power"; lose Imperial Authority and legitimacy (there is nothing we can do), lose gold (his loyalties can be bought), lose stability and gain Imperial Authority (execute the traitor) 
#			+ Ex: "Settle the Veterans", players can either get a free "soldiers estate" building (they shall be granted appropriate land) or increment Imperial Authority (they shall awarded with titles)
#			+ Ex: "Expand the Domestikos", players can spend gold and get a kastron + Imperial Authority, or gain gold and lose Imperial Authority
#			+ Ex: "Peasants upset by military", players lose control in a random province and lose Imperial Authority or get unrest penalty to peasant pops and gain Imperial Authority
#			+ Ex: Events based on laws or reforms;
#				- Exclude the Strategoi should trigger a bunch of negative events based on the generals being out of favor in Constantinople (these should be replaced by good events if you change the laws)
#			+ Ex: Having the Legacy of Andronikos reform should trigger events with the Serbs and Bulgarians
#			+ Ex: Events based on buildings, like good events for having the Pandadikterion, Thema HQs, or lots of burgi
#		* History-based events
#		* Assigning Strategos should have an impact
#			+ More 'bad' events if no Strategos is assigned
#		* Lose Imperial Authority when ruler dies?
#		* 
#	Satisfaction Condition: Imperial Authority = 20
pm_flavor_byz.101 = {
	type = country_event
	title = pm_flavor_byz.101.title
	illustration_tags = {
		10 = exterior
		10 = angry
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { ruler = character:byz_andronikos_iii_palaiologos }
				desc = pm_flavor_byz.101.desc
			}
			triggered_desc = {
				desc = pm_flavor_byz.101.desc.fallback
			}
		}
	}
		
	fire_only_once = yes
	dynamic_historical_event = {
		tag = BYZ
		from = 1336.1.1
		to = 1338.1.1
		monthly_chance = 100
	}
	trigger = {
		OR = {
				byz_ruled_by_palaiologos_trigger = yes
				any_active_disaster = {
					disaster_type = disaster_type:succession_crisis
				}
		}
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		set_variable = { name = pm_byz_imperial_authority_var value = -1 }
		byz_set_localization_scopes = yes
		location:polygyros = {
			add_location_modifier = {
				modifier = imperial_mint
				years = -1
				mode = add_and_extend
			}
			add_location_modifier = {
				modifier = emergency_minting
				years = 50
				mode = add_and_extend
			}
			add_location_modifier = {
				modifier = pm_extensive_mining_complex
				years = -1
				mode = add_and_extend
			}
		} #pm_extensive_mining_complex
		location:constantinople = {
			add_location_modifier = {
				modifier = imperial_looms
				years = -1
				mode = add_and_extend
			}
			add_location_modifier = {
				modifier = imperial_docks
				years = -1
				mode = add_and_extend
			}
		} #pm_extensive_mining_complex
	}
	# Byzantium will prevail!
	option = {
		name = pm_flavor_byz.101.a
		historical_option = yes
		#add_legitimacy = -40
		add_country_modifier = { modifier = byz_imperial_neglect years = -1 mode = add_and_extend }
		custom_tooltip = {
			text = pm_flavor_byz.101.a.at
			every_owned_location = {
				change_control = -0.10
			}
		}
		custom_tooltip = pm_flavor_byz.101.a.bt
		custom_tooltip = pm_flavor_byz.101.a.ct
	}
	option = {
		name = pm_flavor_byz.101.b
		custom_tooltip = pm_flavor_byz.101.b.at
	}
}
